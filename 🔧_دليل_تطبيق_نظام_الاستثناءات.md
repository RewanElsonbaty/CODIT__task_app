```dart
// lib/features/installments/data/mock_installments_api.dart
import '../../../core/errors/exceptions.dart';

class MockInstallmentsApi {
  /// إنشاء عقد جديد
  Future<Map<String, dynamic>> createContract({
    required String customerId,
    required double totalAmount,
    required double downPayment,
    required int installmentsCount,
  }) async {
    try {
      // ✅ التحقق من البيانات
      if (customerId.isEmpty) {
        throw ValidationException('معرف العميل مطلوب');
      }

      if (totalAmount <= 0) {
        throw ValidationException('المبلغ يجب أن يكون أكبر من صفر');
      }

      if (downPayment < 0 || downPayment > totalAmount) {
        throw ValidationException('الدفعة المقدمة غير صحيحة');
      }

      if (installmentsCount <= 0) {
        throw ValidationException('عدد الأقساط يجب أن يكون على الأقل 1');
      }

      // ✅ التحقق من وجود العميل
      final customer = _customers.firstWhere(
        (c) => c['id'] == customerId,
        orElse: () => throw NotFoundException('العميل غير موجود'),
      );

      // ✅ التحقق من الصلاحية الائتمانية
      final creditScore = customer['credit_score'] as int;
      if (creditScore < 600) {
        throw UnauthorizedException('التقييم الائتماني للعميل لا يسمح بالتقسيط');
      }

      // ✅ التحقق من الديون المتأخرة
      final overdueDebt = customer['overdue_debt'] as double;
      if (overdueDebt > 0) {
        throw ConflictException('العميل لديه ديون متأخرة بقيمة ${overdueDebt.toStringAsFixed(2)}');
      }

      // ✅ إنشاء العقد
      final contractId = 'cont_${DateTime.now().millisecondsSinceEpoch}';
      final remainingAmount = totalAmount - downPayment;
      final monthlyInstallment = remainingAmount / installmentsCount;

      final contract = {
        'id': contractId,
        'customer_id': customerId,
        'total_amount': totalAmount,
        'down_payment': downPayment,
        'remaining_amount': remainingAmount,
        'monthly_installment': monthlyInstallment,
        'installments_count': installmentsCount,
        'paid_installments': 0,
        'start_date': DateTime.now().toIso8601String(),
        'status': 'active',
        'created_at': DateTime.now().toIso8601String(),
      };

      _contracts.add(contract);

      return {'success': true, 'data': contract};

    } on ValidationException {
      rethrow;
    } on NotFoundException {
      rethrow;
    } on UnauthorizedException {
      rethrow;
    } on ConflictException {
      rethrow;
    } catch (e) {
      // ❌ أي خطأ آخر يصبح MockNetworkException
      throw MockNetworkException('فشل في إنشاء عقد التقسيط: ${e.toString()}');
    }
  }

  /// الحصول على عقد بمعرفه
  Future<Map<String, dynamic>> getContract(String contractId) async {
    try {
      if (contractId.isEmpty) {
        throw ValidationException('معرف العقد مطلوب');
      }

      await Future.delayed(const Duration(milliseconds: 500));

      final contract = _contracts.firstWhere(
        (c) => c['id'] == contractId,
        orElse: () => throw NotFoundException('العقد غير موجود'),
      );

      return {'success': true, 'data': contract};

    } on ValidationException {
      rethrow;
    } on NotFoundException {
      rethrow;
    } catch (e) {
      throw MockNetworkException('فشل في تحميل بيانات العقد: ${e.toString()}');
    }
  }

  /// تسجيل دفعة
  Future<Map<String, dynamic>> recordPayment({
    required String contractId,
    required double amount,
    required String paymentMethod,
  }) async {
    try {
      // التحقق
      if (contractId.isEmpty) {
        throw ValidationException('معرف العقد مطلوب');
      }

      if (amount <= 0) {
        throw ValidationException('المبلغ يجب أن يكون أكبر من صفر');
      }

      if (paymentMethod.isEmpty) {
        throw ValidationException('طريقة الدفع مطلوبة');
      }

      // البحث عن العقد
      final contractIndex = _contracts.indexWhere((c) => c['id'] == contractId);
      if (contractIndex == -1) {
        throw NotFoundException('العقد غير موجود');
      }

      final contract = _contracts[contractIndex];
      final remainingAmount = contract['remaining_amount'] as double;

      // التحقق من المبلغ
      if (amount > remainingAmount) {
        throw ValidationException(
          'المبلغ المدفوع (${amount.toStringAsFixed(2)}) '
          'أكبر من المبلغ المتبقي (${remainingAmount.toStringAsFixed(2)})'
        );
      }

      // تسجيل الدفعة
      final paymentId = 'pay_${DateTime.now().millisecondsSinceEpoch}';
      final payment = {
        'id': paymentId,
        'contract_id': contractId,
        'amount': amount,
        'payment_method': paymentMethod,
        'payment_date': DateTime.now().toIso8601String(),
      };

      _payments.add(payment);

      // تحديث العقد
      final newRemaining = remainingAmount - amount;
      final paidInstallments = contract['paid_installments'] as int;
      
      _contracts[contractIndex] = {
        ...contract,
        'remaining_amount': newRemaining,
        'paid_installments': paidInstallments + 1,
        'status': newRemaining == 0 ? 'completed' : 'active',
      };

      return {'success': true, 'data': payment};

    } on ValidationException {
      rethrow;
    } on NotFoundException {
      rethrow;
    } catch (e) {
      throw MockNetworkException('فشل في تسجيل الدفعة: ${e.toString()}');
    }
  }
}
```

```dart
// lib/features/installments/data/repository/installments_repository_impl.dart
import 'package:dartz/dartz.dart';
import '../../../../core/errors/failures.dart';
import '../../../../core/errors/exceptions.dart';
import '../../domain/models/contract.dart';
import '../../domain/models/payment.dart';
import '../../domain/repository/installments_repository.dart';
import '../mock_installments_api.dart';

class InstallmentsRepositoryImpl implements InstallmentsRepository {
  final MockInstallmentsApi _api;

  InstallmentsRepositoryImpl(this._api);

  @override
  Future<Either<Failure, Contract>> createContract({
    required String customerId,
    required double totalAmount,
    required double downPayment,
    required int installmentsCount,
  }) async {
    try {
      final result = await _api.createContract(
        customerId: customerId,
        totalAmount: totalAmount,
        downPayment: downPayment,
        installmentsCount: installmentsCount,
      );

      final contract = Contract.fromJson(result['data']);
      return Right(contract);

    } on ValidationException catch (e) {
      return Left(ValidationFailure(message: e.message));
    } on NotFoundException catch (e) {
      return Left(NotFoundFailure(message: e.message));
    } on UnauthorizedException catch (e) {
      return Left(UnauthorizedFailure(message: e.message));
    } on ConflictException catch (e) {
      return Left(ConflictFailure(message: e.message));
    } on MockNetworkException catch (e) {
      return Left(NetworkFailure(message: e.message));
    } catch (e) {
      return Left(ServerFailure(message: 'حدث خطأ غير متوقع'));
    }
  }

  @override
  Future<Either<Failure, Contract>> getContract(String contractId) async {
    try {
      final result = await _api.getContract(contractId);
      final contract = Contract.fromJson(result['data']);
      return Right(contract);

    } on ValidationException catch (e) {
      return Left(ValidationFailure(message: e.message));
    } on NotFoundException catch (e) {
      return Left(NotFoundFailure(message: e.message));
    } on MockNetworkException catch (e) {
      return Left(NetworkFailure(message: e.message));
    } catch (e) {
      return Left(ServerFailure(message: 'حدث خطأ غير متوقع'));
    }
  }

  @override
  Future<Either<Failure, Payment>> recordPayment({
    required String contractId,
    required double amount,
    required String paymentMethod,
  }) async {
    try {
      final result = await _api.recordPayment(
        contractId: contractId,
        amount: amount,
        paymentMethod: paymentMethod,
      );

      final payment = Payment.fromJson(result['data']);
      return Right(payment);

    } on ValidationException catch (e) {
      return Left(ValidationFailure(message: e.message));
    } on NotFoundException catch (e) {
      return Left(NotFoundFailure(message: e.message));
    } on MockNetworkException catch (e) {
      return Left(NetworkFailure(message: e.message));
    } catch (e) {
      return Left(ServerFailure(message: 'حدث خطأ غير متوقع'));
    }
  }
}
```

```dart
// lib/features/installments/presentation/cubit/installments_cubit.dart
import 'package:flutter_bloc/flutter_bloc.dart';
import '../../domain/repository/installments_repository.dart';
import '../../domain/models/contract.dart';
import '../../domain/models/payment.dart';

class InstallmentsCubit extends Cubit<InstallmentsState> {
  final InstallmentsRepository _repository;

  InstallmentsCubit(this._repository) : super(InstallmentsInitial());

  /// إنشاء عقد جديد
  Future<void> createContract({
    required String customerId,
    required double totalAmount,
    required double downPayment,
    required int installmentsCount,
  }) async {
    emit(InstallmentsLoading());

    final result = await _repository.createContract(
      customerId: customerId,
      totalAmount: totalAmount,
      downPayment: downPayment,
      installmentsCount: installmentsCount,
    );

    result.fold(
      (failure) => emit(InstallmentsError(message: failure.message)),
      (contract) => emit(ContractCreated(contract: contract)),
    );
  }

  /// تحميل عقد
  Future<void> loadContract(String contractId) async {
    emit(InstallmentsLoading());

    final result = await _repository.getContract(contractId);

    result.fold(
      (failure) => emit(InstallmentsError(message: failure.message)),
      (contract) => emit(ContractLoaded(contract: contract)),
    );
  }

  /// تسجيل دفعة
  Future<void> recordPayment({
    required String contractId,
    required double amount,
    required String paymentMethod,
  }) async {
    emit(PaymentProcessing());

    final result = await _repository.recordPayment(
      contractId: contractId,
      amount: amount,
      paymentMethod: paymentMethod,
    );

    result.fold(
      (failure) => emit(InstallmentsError(message: failure.message)),
      (payment) => emit(PaymentRecorded(payment: payment)),
    );
  }
}

// States
abstract class InstallmentsState {}

class InstallmentsInitial extends InstallmentsState {}

class InstallmentsLoading extends InstallmentsState {}

class PaymentProcessing extends InstallmentsState {}

class ContractCreated extends InstallmentsState {
  final Contract contract;
  ContractCreated({required this.contract});
}

class ContractLoaded extends InstallmentsState {
  final Contract contract;
  ContractLoaded({required this.contract});
}

class PaymentRecorded extends InstallmentsState {
  final Payment payment;
  PaymentRecorded({required this.payment});
}

class InstallmentsError extends InstallmentsState {
  final String message;
  InstallmentsError({required this.message});
}
```

### 4. UI Layer (مع ExceptionHandler)

```dart
// lib/features/installments/presentation/screens/create_contract_screen.dart
import 'package:flutter/material.dart';
import 'package:flutter_bloc/flutter_bloc.dart';
import '../../../../core/utils/exception_handler.dart';
import '../cubit/installments_cubit.dart';

class CreateContractScreen extends StatefulWidget {
  const CreateContractScreen({super.key});

  @override
  State<CreateContractScreen> createState() => _CreateContractScreenState();
}

class _CreateContractScreenState extends State<CreateContractScreen> {
  final _formKey = GlobalKey<FormState>();
  final _customerIdController = TextEditingController();
  final _totalAmountController = TextEditingController();
  final _downPaymentController = TextEditingController();
  final _installmentsCountController = TextEditingController();

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('إنشاء عقد تقسيط جديد')),
      body: BlocConsumer<InstallmentsCubit, InstallmentsState>(
        listener: (context, state) {
          if (state is ContractCreated) {
            // ✅ Success
            ExceptionHandler.showSuccessSnackBar(
              context,
              'تم إنشاء العقد رقم ${state.contract.id} بنجاح',
            );
            Navigator.pop(context, state.contract);
          } else if (state is InstallmentsError) {
            // ❌ Error
            ExceptionHandler.showErrorSnackBar(context, state.message);
          }
        },
        builder: (context, state) {
          final isLoading = state is InstallmentsLoading;

          return Form(
            key: _formKey,
            child: ListView(
              padding: const EdgeInsets.all(16),
              children: [
                // Form fields...
                TextFormField(
                  controller: _customerIdController,
                  decoration: const InputDecoration(
                    labelText: 'معرف العميل',
                  ),
                  validator: (value) {
                    if (value == null || value.isEmpty) {
                      return 'الرجاء إدخال معرف العميل';
                    }
                    return null;
                  },
                ),
                const SizedBox(height: 16),

                TextFormField(
                  controller: _totalAmountController,
                  decoration: const InputDecoration(
                    labelText: 'المبلغ الإجمالي',
                  ),
                  keyboardType: TextInputType.number,
                  validator: (value) {
                    if (value == null || value.isEmpty) {
                      return 'الرجاء إدخال المبلغ';
                    }
                    final amount = double.tryParse(value);
                    if (amount == null || amount <= 0) {
                      return 'المبلغ غير صحيح';
                    }
                    return null;
                  },
                ),
                const SizedBox(height: 16),

                TextFormField(
                  controller: _downPaymentController,
                  decoration: const InputDecoration(
                    labelText: 'الدفعة المقدمة',
                  ),
                  keyboardType: TextInputType.number,
                  validator: (value) {
                    if (value == null || value.isEmpty) {
                      return 'الرجاء إدخال الدفعة المقدمة';
                    }
                    final amount = double.tryParse(value);
                    if (amount == null || amount < 0) {
                      return 'المبلغ غير صحيح';
                    }
                    return null;
                  },
                ),
                const SizedBox(height: 16),

                TextFormField(
                  controller: _installmentsCountController,
                  decoration: const InputDecoration(
                    labelText: 'عدد الأقساط',
                  ),
                  keyboardType: TextInputType.number,
                  validator: (value) {
                    if (value == null || value.isEmpty) {
                      return 'الرجاء إدخال عدد الأقساط';
                    }
                    final count = int.tryParse(value);
                    if (count == null || count <= 0) {
                      return 'العدد غير صحيح';
                    }
                    return null;
                  },
                ),
                const SizedBox(height: 32),

                ElevatedButton(
                  onPressed: isLoading ? null : _handleSubmit,
                  child: isLoading
                      ? const SizedBox(
                          height: 20,
                          width: 20,
                          child: CircularProgressIndicator(strokeWidth: 2),
                        )
                      : const Text('إنشاء العقد'),
                ),
              ],
            ),
          );
        },
      ),
    );
  }

  void _handleSubmit() {
    if (_formKey.currentState!.validate()) {
      // ⚠️ تحذير قبل الإرسال
      final totalAmount = double.parse(_totalAmountController.text);
      final downPayment = double.parse(_downPaymentController.text);
      
      if (downPayment > totalAmount) {
        ExceptionHandler.showWarningSnackBar(
          context,
          'الدفعة المقدمة أكبر من المبلغ الإجمالي',
        );
        return;
      }

      // ℹ️ معلومة
      ExceptionHandler.showInfoSnackBar(
        context,
        'جاري إنشاء العقد...',
      );

      context.read<InstallmentsCubit>().createContract(
        customerId: _customerIdController.text.trim(),
        totalAmount: totalAmount,
        downPayment: downPayment,
        installmentsCount: int.parse(_installmentsCountController.text),
      );
    }
  }

  @override
  void dispose() {
    _customerIdController.dispose();
    _totalAmountController.dispose();
    _downPaymentController.dispose();
    _installmentsCountController.dispose();
    super.dispose();
  }
}
```
